class FakeEmbeddingsEndpoint:
    def __init__(self, dimension: int = 2) -> None:
        self.calls: list[list[str]] = []
        self.dimension = dimension

    def create(self, *, model, input):
        texts = list(input)
        self.calls.append(texts)
        return SimpleNamespace(
            data=[
                SimpleNamespace(
                    index=index,
                    embedding=[float(index + 1)] + [1.0] * (self.dimension - 1),
                )
                for index, _text in enumerate(texts)
            ],
            usage=SimpleNamespace(prompt_tokens=len(texts)),
        )


class M0OpenAITests(unittest.TestCase):
    def patched_agent_initialization(self):
        return [
            patch.object(agent_module.Agent, "_build_fts5_index"),
            patch.object(agent_module.Agent, "_build_category_index"),
            patch.object(agent_module.Agent, "_build_vector_index"),
            patch.object(agent_module.BaselineAgent, "_build_index"),
        ]

    def test_canonical_default_is_openai_and_does_not_initialize_bge(self):
        patchers = self.patched_agent_initialization()
        with patch.object(agent_module, "BGEEmbeddingBackend") as bge_constructor:
            for patcher in patchers:
                patcher.start()
            try:
                instance = agent_module.Agent("unused-catalog.jsonl")
                self.assertIsInstance(instance.embedding_backend, OpenAIEmbeddingBackend)
                self.assertEqual(instance.embedding_space_id, OPENAI_EMBEDDING_SPACE_ID)
                bge_constructor.assert_not_called()
            finally:
                instance.connection.close()
                instance.baseline_agent.connection.close()
                for patcher in reversed(patchers):
                    patcher.stop()

    def test_explicit_bge_variant_remains_lazy_and_selectable(self):
        patchers = self.patched_agent_initialization()
        for patcher in patchers:
            patcher.start()
        try:
            with patch.object(agent_bge, "SentenceTransformer") as model_constructor:
                instance = agent_bge.Agent("unused-catalog.jsonl")
                self.assertIsInstance(instance.embedding_backend, BGEEmbeddingBackend)
                self.assertIsNone(instance.embedding_backend._model)
                model_constructor.assert_not_called()
        finally:
            instance.connection.close()
            instance.baseline_agent.connection.close()
            for patcher in reversed(patchers):
                patcher.stop()

    def test_m0_catalog_and_query_operations_share_one_embedding_space(self):
        endpoint = FakeEmbeddingsEndpoint(dimension=3072)
        backend = OpenAIEmbeddingBackend(client=SimpleNamespace(embeddings=endpoint))
        catalog_vectors = backend.embed_catalog(["first product", "second product"])

        instance = agent_module.Agent.__new__(agent_module.Agent)
        instance.embedding_backend = backend
        instance.embedding_space_id = backend.embedding_space_id
        instance.catalog_embeddings = catalog_vectors
        instance.catalog_ids = ["first", "second"]
        instance.instrumentation = {"semantic_queries": []}
        instance._dense_retrieve("walking boots", top_n=2)

        self.assertEqual(backend.embedding_space_id, OPENAI_EMBEDDING_SPACE_ID)
        self.assertEqual(
            instance.instrumentation["semantic_queries"][0]["embedding_space_id"],
            OPENAI_EMBEDDING_SPACE_ID,
        )
        self.assertEqual(endpoint.calls, [["first product", "second product"], ["walking boots"]])

    def test_m0_config_disables_slow_memory_ranking_while_phase5_hook_is_shadow_only(self):
        config = run_m0.load_config()
        self.assertEqual(config["embedding"]["dimensions"], 3072)
        self.assertTrue(config["embedding"]["normalized"])
        self.assertFalse(config["embedding"]["fallback_enabled"])
        self.assertTrue(config["memory"]["fast_memory_enabled"])
        self.assertFalse(config["memory"]["slow_memory_enabled"])
        for name in ("slow_memory", "historical_sessions", "rerank_with_memory"):
            self.assertFalse(hasattr(agent_module.Agent, name))
        self.assertTrue(hasattr(agent_module.Agent, "end_session"))
        self.assertTrue(hasattr(agent_module.Agent, "get_visible_memories"))

    def test_importing_m0_runner_does_not_construct_openai_client(self):
        with patch("openai.OpenAI") as client_constructor:
            importlib.reload(run_m0)
        client_constructor.assert_not_called()

    def test_mocked_m0_run_writes_required_artifacts(self):
        telemetry = {
            "turns": [
                {"route": "fast", "respond_seconds": 0.1},
                {"route": "full", "respond_seconds": 0.2},
            ],
            "semantic_queries": [
                {
                    "query_embedding_seconds": 0.03,
                    "dense_search_seconds": 0.01,
                }
            ],
            "agent_errors": [],
            "baseline_fallback_count": 0,
            "embedding_space_id": OPENAI_EMBEDDING_SPACE_ID,
            "initialization": {
                "cache_status": "hit",
                "catalog_embedding_generation_seconds": 0.0,
            },
            "embedding_api": {
                "request_count": 1,
                "input_tokens": 4,
                "request_latencies_seconds": [0.03],
            },
        }
        fake_agent = MagicMock()
        fake_agent.embedding_space_id = OPENAI_EMBEDDING_SPACE_ID
        fake_agent.get_instrumentation.return_value = telemetry
        evaluator_metrics = {
            "sample_count": 1,
            "hit_rate_at_10": 1.0,
            "mrr": 1.0,
            "mttc": 1.0,
            "scenario_metrics": {"buying": {"sample_count": 1}},
        }

        with tempfile.TemporaryDirectory() as directory:
            destination = Path(directory)
            with (
                patch.object(run_m0, "RESULTS_DIR", destination),
                patch.object(run_m0, "Agent", return_value=fake_agent),
                patch.object(run_m0, "load_evaluator_data", return_value=([{}], ["a"], {}, {})),
                patch.object(run_m0, "run_test_preflight", return_value={"status": "passed", "test_count": 33}),
                patch.object(run_m0, "require_openai_key"),
                patch("experiment_1.run_eval_v2.evaluate_v2", return_value=evaluator_metrics),
            ):
                run_m0.run_full(allow_catalog_build=False)

            for name in ("config.json", "metrics.json", "latency.json", "run_summary.md", "FREEZE.md"):
                self.assertTrue((destination / name).exists(), name)
            saved_config = json.loads((destination / "config.json").read_text(encoding="utf-8"))
            self.assertEqual(saved_config["baseline_name"], "M0_OPENAI")
            self.assertFalse(saved_config["memory"]["slow_memory_enabled"])


if __name__ == "__main__":
    unittest.main()
